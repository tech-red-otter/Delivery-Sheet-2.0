SELECT expe.customer,bc.contact_name, expe.products FROM delivery_export."exceptions" expe
LEFT JOIN data_imports_n8n.books_contacts bc on expe.customer = bc.contact_id
WHERE bc.contact_name ilike '%{{data_table.searchText || ""}}%'
ORDER BY "{{data_table.sortOrder.column || 'customer'}}" {{data_table.sortOrder.order || 'ASC'}}
LIMIT {{data_table.pageSize}}
OFFSET {{(data_table.pageNo - 1) * data_table.pageSize}};
