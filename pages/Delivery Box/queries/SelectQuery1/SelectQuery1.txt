WITH master_box_aggregates AS (
    SELECT
        db_inner.master_box,
        COUNT(*) AS total_records_as_master_boxes,
        ARRAY_AGG(DISTINCT bc_inner.contact_name) AS customers
    FROM
        delivery.delivery_box db_inner
    LEFT JOIN
        delivery.sales_orders_map som_inner
    ON
        db_inner.salesorder_id = som_inner.salesorder_id
    LEFT JOIN
        public.customers bc_inner
    ON
        som_inner.customer_id = bc_inner.customer_id
    WHERE
        db_inner.archived != true
        AND som_inner.date = '{{shipment_date.formattedDate}}'
    GROUP BY
        db_inner.master_box
)
SELECT
    db.id,
    db.box_code,
    db.packaging_details,
    db.products,
    bc.contact_name,
    som.make_your_own,
    som.customer_id,
    COALESCE(mb.total_records_as_master_boxes, 1) AS total_records_as_master_boxes,
    COALESCE(mb.customers, ARRAY[bc.contact_name]) AS customers
FROM
    delivery.delivery_box db
LEFT JOIN
    delivery.sales_orders_map som
ON
    db.salesorder_id = som.salesorder_id
LEFT JOIN
    public.customers bc
ON
    som.customer_id = bc.customer_id
LEFT JOIN
    master_box_aggregates mb
ON
    db.id = mb.master_box -- Ensure correct data types for this join
WHERE
    db.box_code ILIKE '%{{data_table.searchText || ""}}%'
    AND db.master_box IS NULL 
    AND db.shipment_date = '{{shipment_date.formattedDate}}'
    AND db.box_code != 'BLANK_BOX'
    AND db.archived != true
ORDER BY
    "{{data_table.sortOrder.column || 'box_code'}}" {{data_table.sortOrder.order || 'ASC'}}
LIMIT
    {{data_table.pageSize}}
OFFSET
    {{(data_table.pageNo - 1) * data_table.pageSize}};
