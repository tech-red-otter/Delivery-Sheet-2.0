SELECT * FROM delivery.master_box
WHERE box_code ilike '%{{data_table.searchText || ""}}%' 
  AND (
    '{{dl_curated_box.selectedOptionValue || ""}}' = '' 
    OR '{{dl_curated_box.selectedOptionValue}}' = ANY(curated_box_name)
  )
ORDER BY "{{data_table.sortOrder.column || 'box_code'}}" {{data_table.sortOrder.order || 'ASC'}}
LIMIT {{data_table.pageSize}}
OFFSET {{(data_table.pageNo - 1) * data_table.pageSize}};
