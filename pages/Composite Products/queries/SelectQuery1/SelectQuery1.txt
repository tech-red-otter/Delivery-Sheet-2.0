SELECT cp.*, ivi.name FROM delivery.composite_products cp
LEFT JOIN delivery.inv_items ivi on cp.composite_product = ivi.sku 
WHERE "name" ilike '%{{data_table.searchText || ""}}%'
ORDER BY "{{data_table.sortOrder.column || 'composite_product'}}" {{data_table.sortOrder.order || 'ASC'}}
LIMIT {{data_table.pageSize}}
OFFSET {{(data_table.pageNo - 1) * data_table.pageSize}};
