SELECT m.salesorder_id,m.salesorder_number,m.date,m.customer_id,c.contact_name
FROM delivery.sales_orders_map m inner join public.customers c on c.customer_id=m.customer_id
WHERE m.salesorder_id not in(select salesorder_id from public.salesorder_delivery) 
AND m.date>='2026-01-01' and m.date<=now() order by date desc
