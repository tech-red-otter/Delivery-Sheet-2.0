SELECT 
    som.*,
    bc.contact_name, 
    db.box_code,
    (SELECT products 
     FROM delivery.exceptions 
     WHERE customer = som.customer_id) AS exceptions,
    (SELECT box_code 
     FROM delivery.delivery_box
     WHERE id = db.master_box) AS master_box,
    (SELECT json_agg(isli.name)
     FROM zoho_records.inv_salesorders_list_item isli 
     WHERE som.salesorder_id = isli.salesorder_id) AS sales_order_items,
    COUNT(*) OVER() AS total_count
FROM delivery.sales_orders_map som
LEFT JOIN public.customers bc  ON som.customer_id = bc.customer_id
LEFT JOIN delivery.delivery_box db ON som.delivery_box::uuid = db.id
WHERE {{ shipment_date.formattedDate ? ` som.date = '${shipment_date.formattedDate}'` : "" }}
    AND (
        '{{order_item.text || ""}}' = ''
        OR EXISTS (
            SELECT 1
            FROM zoho_records.inv_salesorders_list_item isli
            WHERE som.salesorder_id = isli.salesorder_id
              AND isli.name ILIKE '%{{order_item.text || ""}}%'
        )
    )
		{{ only_blank_box.isChecked ? " AND db.box_code = 'BLANK_BOX'" : "" }}
		{{ only_non_myo_box.isChecked? " AND db.box_code not ilike 'MYO-'":"" }}
    AND som.archived IS DISTINCT FROM TRUE
ORDER BY som.salesorder_number ASC
