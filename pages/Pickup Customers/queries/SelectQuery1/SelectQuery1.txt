SELECT * FROM (
  SELECT pc.*, so.customer_name,
  COUNT(*) OVER() AS total_count
  FROM delivery.pickup_customers pc
  LEFT JOIN LATERAL (
    SELECT customer_name
    FROM public.salesorders
    WHERE customer_id = pc.id
    LIMIT 1
  ) so ON true
) sub
WHERE sub."id" ilike '%{{data_table.searchText || ""}}%'
   OR sub."customer_name" ilike '%{{data_table.searchText || ""}}%'
LIMIT {{data_table.pageSize}}
OFFSET {{(data_table.pageNo - 1) * data_table.pageSize}};
